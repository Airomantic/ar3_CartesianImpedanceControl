cmake_minimum_required(VERSION 3.16)
project(ar3_cartesian_impedance)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Threads REQUIRED)

# Deliberately isolated from the older rokae_hardware/sdk ABI.
set(AR3_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../xCoreSDK-0.7.1.ar_10/librokae-v0.7.1.ar_10"
  CACHE PATH "xCoreSDK 0.7.1 AR directory")
set(AR3_SDK_LIB "${AR3_SDK_ROOT}/lib/Linux/${CMAKE_SYSTEM_PROCESSOR}")
if(NOT EXISTS "${AR3_SDK_LIB}/libxCoreSDK.a")
  message(FATAL_ERROR "Missing AR SDK library: ${AR3_SDK_LIB}/libxCoreSDK.a")
endif()
add_executable(cartesian_impedance_node src/cartesian_impedance_rt_node.cpp)
target_include_directories(cartesian_impedance_node PRIVATE include "${AR3_SDK_ROOT}/include")
target_compile_options(cartesian_impedance_node PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(cartesian_impedance_node rclcpp std_msgs geometry_msgs std_srvs)
target_link_libraries(cartesian_impedance_node
  "${AR3_SDK_LIB}/libxCoreSDK.a" "${AR3_SDK_LIB}/libxMateModel.a"
  Threads::Threads Eigen3::Eigen ${CMAKE_DL_LIBS})
install(TARGETS cartesian_impedance_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  enable_testing()
  add_executable(command_validation_test test/command_validation_test.cpp)
  target_include_directories(command_validation_test PRIVATE include)
  target_link_libraries(command_validation_test Eigen3::Eigen)
  add_test(NAME command_validation COMMAND command_validation_test)

  add_executable(load_config_test test/load_config_test.cpp)
  target_include_directories(load_config_test PRIVATE include)
  target_compile_options(load_config_test PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME load_config COMMAND load_config_test)

  add_executable(wrench_guard_test test/wrench_guard_test.cpp)
  target_include_directories(wrench_guard_test PRIVATE include)
  target_compile_options(wrench_guard_test PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(wrench_guard_test Eigen3::Eigen)
  add_test(NAME wrench_guard COMMAND wrench_guard_test)

  add_executable(rt_startup_recovery_test test/rt_startup_recovery_test.cpp)
  target_include_directories(rt_startup_recovery_test PRIVATE include)
  target_compile_options(rt_startup_recovery_test PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME rt_startup_recovery COMMAND rt_startup_recovery_test)

  add_executable(rt_core_test test/rt_core_test.cpp)
  target_include_directories(rt_core_test PRIVATE include)
  target_compile_options(rt_core_test PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(rt_core_test Eigen3::Eigen Threads::Threads)
  add_test(NAME rt_core COMMAND rt_core_test)
  set_tests_properties(rt_core PROPERTIES TIMEOUT 20)
  option(AR3_SANITIZE_RT_TESTS "Run offline RT core regressions under AddressSanitizer" OFF)
  if(AR3_SANITIZE_RT_TESTS)
    foreach(test_target command_validation_test load_config_test wrench_guard_test rt_core_test rt_startup_recovery_test)
      target_compile_options(${test_target} PRIVATE -fsanitize=address -fno-omit-frame-pointer)
      target_link_options(${test_target} PRIVATE -fsanitize=address)
    endforeach()
  endif()
endif()
ament_package()
